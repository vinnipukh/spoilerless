```python
# spoilerless/app/main.py
def _trusted_hosts() -> list[str]:
    """Derive allowed host header values for TrustedHostMiddleware.
    
    Uses settings.allowed_hosts when populated; falls back to standard
    local and production backend domains (including *.onrender.com).
    """
    try:
        settings = get_settings()
    except Exception:
        return ["localhost", "127.0.0.1", "api.spoilerless.net", "*.onrender.com", "testserver"]
    if settings.allowed_hosts.strip():
        return [h.strip() for h in settings.allowed_hosts.split(",") if h.strip()]
    return [
        "localhost",
        "127.0.0.1",
        "api.spoilerless.net",
        "*.onrender.com",
        "testserver",
    ]
```
