import json

# --- Helper factories ---

def make_claim(
    subject_id: str = "character:dexter",
    predicate: str = "conceals",
    object_id: str = "object:blood_slides",
    evidence_text: str = "Dexter hides his blood slides.",
    **overrides,
) -> dict:
    base = {
        "schema_version": "0.1",
        "subject_id": subject_id,
        "predicate": predicate,
        "object_id": object_id,
        "claim_type": "character_relationship",
        "confidence_level": "medium",
        "relationship_effect": "neutral",
        "visible_from_order": 1,
        "valid_from_order": 1,
        "valid_until_order": None,
        "evidence_text": evidence_text,
        "evidence_locator": "S01E01:10:00",
        "source_type": "transcript",
        "source_locator": "https://example.com",
        "episode_id": "episode:dexter:s01e01",
    }
    base.update(overrides)
    return base


class TestExtractionClaim:
    """PREP-01: Extraction claim schema validation."""

    def test_valid_claim(self):
        """A correctly structured claim passes validation."""
        claim = ExtractionClaim(**make_claim())
        assert claim.schema_version == "0.1"
        assert claim.claim_type == "character_relationship"
        assert claim.candidate_id.startswith("extracted:")
        assert len(claim.candidate_id) == 16 + len("extracted:")  # extracted: + 16 hex chars

    def test_extra_fields_forbidden(self):
        """Extra fields beyond the schema are rejected (D-04)."""
        data = make_claim(extra_field="should_not_exist")
        with pytest.raises(ValidationError, match="extra"):
            ExtractionClaim(**data)

    def test_ontology_claim_type_validation(self):
        """Claim type must be in the ontology allowlist (D-04)."""
        data = make_claim(claim_type="invalid_claim_type")
        with pytest.raises(ValidationError, match="claim_type"):
            ExtractionClaim(**data)

    def test_ontology_confidence_level_validation(self):
        """Confidence level must be in the ontology allowlist."""
        data = make_claim(confidence_level="extremely_high")
        with pytest.raises(ValidationError, match="confidence"):
            ExtractionClaim(**data)

    def test_ontology_relationship_effect_validation(self):
        """Relationship effect must be in the ontology allowlist."""
        data = make_claim(relationship_effect="super_effect")
        with pytest.raises(ValidationError, match="relationship"):
            ExtractionClaim(**data)

    def test_deterministic_candidate_id(self):
        """Same payload produces same candidate ID (D-11)."""
        c1 = ExtractionClaim(**make_claim())
        c2 = ExtractionClaim(**make_claim())
        assert c1.candidate_id == c2.candidate_id

    def test_different_payload_different_id(self):
        """Different payloads produce different candidate IDs."""
        c1 = ExtractionClaim(**make_claim(evidence_text="First evidence."))
        c2 = ExtractionClaim(**make_claim(evidence_text="Different evidence."))
        assert c1.candidate_id != c2.candidate_id

    def test_confidence_defaults_to_medium(self):
        """confidence_level defaults to 'medium'."""
        data = make_claim()
        del data["confidence_level"]
        claim = ExtractionClaim(**data)
        assert claim.confidence_level == "medium"

    def test_schema_version_default(self):
        """schema_version defaults to '0.1'."""
        claim = ExtractionClaim(**make_claim())
        assert claim.schema_version == "0.1"

        graph1 = graph_r1.json()
        # Find S01E01-visible claims
        claims1 = [c for c in graph1.get("claims", []) if c.get("origin") == "candidate"]
        # At least the S01E01 claims should be visible

        # Graph at boundary=2
        graph_r2 = seeded_client.get(
            f"/api/series/{self.SERIES_ID}/graph?visible_until_order=2",
        )
        assert graph_r2.status_code == 200
        graph2 = graph_r2.json()
        claims2 = [c for c in graph2.get("claims", []) if c.get("origin") == "candidate"]

        # Boundary=2 should include more claims than boundary=1
        assert len(claims2) >= len(claims1), "Higher boundary should include at least as many claims"

    def test_future_candidates_not_leaked(self, seeded_client: TestClient, extraction_fixture: dict):

            f"/api/series/{self.SERIES_ID}/candidates/ingest",
            json=extraction_fixture,

        list_r1 = seeded_client.get(
            f"/api/series/{self.SERIES_ID}/candidates?visible_until_order=1",

        at_boundary_1 = list_r1.json()

        list_all = seeded_client.get(
            f"/api/series/{self.SERIES_ID}/candidates",
        )

            c["id"]

            if c["visible_from_order"] > 1

        at_1_ids = {c["id"] for c in at_boundary_1 if c.get("origin") == "candidate"}
        assert invisible_ids.isdisjoint(at_1_ids), (
            f"Invisible claims leaked at boundary=1: {invisible_ids & at_1_ids}"
