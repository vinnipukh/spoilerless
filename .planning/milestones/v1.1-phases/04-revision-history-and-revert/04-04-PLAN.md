```typescript
import { apiFetch } from './client'
import type { RevisionResponse } from '../types/revision'

export function getRevisions(
  seriesId: string,
  visibleUntilOrder: number,
  resourceType?: string,
  resourceId?: string,
): Promise<RevisionResponse[]> {
  let url = `/api/series/${encodeURIComponent(seriesId)}/revisions?visible_until_order=${visibleUntilOrder}`
  if (resourceType) url += `&resource_type=${encodeURIComponent(resourceType)}`
  if (resourceId) url += `&resource_id=${encodeURIComponent(resourceId)}`
  return apiFetch<RevisionResponse[]>(url)
}

export function getRevision(
  seriesId: string,
  revisionId: string,
  visibleUntilOrder: number,
): Promise<RevisionResponse> {
  return apiFetch<RevisionResponse>(
    `/api/series/${encodeURIComponent(seriesId)}/revisions/${encodeURIComponent(revisionId)}?visible_until_order=${visibleUntilOrder}`,
  )
}

export function revertRevision(
  seriesId: string,
  revisionId: string,
  visibleUntilOrder: number,
): Promise<RevisionResponse> {
  return apiFetch<RevisionResponse>(
    `/api/series/${encodeURIComponent(seriesId)}/revisions/${encodeURIComponent(revisionId)}/revert?visible_until_order=${visibleUntilOrder}`,
    { method: 'POST' },
  )
}
```

Key decisions:
- `encodeURIComponent` for all dynamic path segments (seriesId, revisionId) — consistent with userContent.ts pattern.
- `revertRevision` uses POST with no body — the backend receives all params via query string (visible_until_order) + path params.
- Returns `Promise<RevisionResponse>` — the backend returns the new REVERTED revision.

Create `frontend/src/hooks/useRevisions.ts` following the useNotes pattern:

```typescript
import { useCallback, useEffect, useRef, useState } from 'react'
import { ApiError } from '../api/client'
import { getRevisions } from '../api/revisions'
import type { RevisionResponse } from '../types/revision'

type State =
  | { status: 'idle' }
  | { status: 'loading' }
  | { status: 'error'; error: ApiError }
  | { status: 'success'; data: RevisionResponse[] }

type Props = {
  seriesId: string | null
  visibleUntilOrder: number | null
  resourceType?: string
  resourceId?: string
}

export function useRevisions({ seriesId, visibleUntilOrder, resourceType, resourceId }: Props) {
  const [state, setState] = useState<State>(() =>
    seriesId && visibleUntilOrder != null ? { status: 'loading' } : { status: 'idle' },
  )

  const key = `${seriesId ?? ''}:${visibleUntilOrder ?? ''}:${resourceType ?? ''}:${resourceId ?? ''}`
  const [prevKey, setPrevKey] = useState(key)
  if (prevKey !== key) {
    setPrevKey(key)
    setState(seriesId && visibleUntilOrder != null ? { status: 'loading' } : { status: 'idle' })
  }

  const fetchKeyRef = useRef(key)
  fetchKeyRef.current = key

  const fetchRevisions = useCallback(() => {
    if (!seriesId || visibleUntilOrder == null) return
    getRevisions(seriesId, visibleUntilOrder, resourceType, resourceId)
      .then((data) => {
        if (fetchKeyRef.current === key) {
          setState({ status: 'success', data })
        }
      })
      .catch((error) => {
        if (fetchKeyRef.current === key) {
          setState({
            status: 'error',
            error: error instanceof ApiError ? error : new ApiError({ code: 'unknown_error', message: 'Request failed.' }),
          })
        }
      })
  }, [seriesId, visibleUntilOrder, resourceType, resourceId, key])

  useEffect(() => {
    if (!seriesId || visibleUntilOrder == null) return
    fetchRevisions()
  }, [fetchRevisions, seriesId, visibleUntilOrder])

  return { ...state, refetch: fetchRevisions }
}
```

Key decisions:
- Uses same `fetchKeyRef` pattern as useNotes to prevent stale state writes after key change.
- Exposes `refetch()` so the History tab can reload after a revert.
- No create/update/delete methods (unlike useNotes) — revisions are append-only from the frontend perspective.
</action>
